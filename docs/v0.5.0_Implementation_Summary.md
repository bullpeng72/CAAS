## 1. BaseValidator 추상 클래스 (NEW)

### 📁 파일: `caas_framework/validation/base_validator.py`

### 목적: 중복 코드 제거

기존 6개 Validator 클래스가 반복하던 패턴을 통합:
- ValidationIssue 생성 (일관된 severity, message, auto_fix)
- ValidationReport 생성 (passed, issues, score 계산)
- 리포트 포맷팅 (human-readable)
- 점수 계산 로직 (critical: -3.0, error: -2.0, warning: -1.0)

### 핵심 기능

```python
class BaseValidator(ABC):
    """
    모든 Validator의 추상 기반 클래스.

    제공 기능:
    - create_issue(): ValidationIssue 생성 헬퍼
    - create_report(): ValidationReport 생성 (자동 점수 계산)
    - calculate_score(): 일관된 점수 계산 (0.0-10.0)
    - format_report(): 인간 친화적 보고서 포맷팅
    """

    @abstractmethod
    def validate(self, *args, **kwargs) -> ValidationReport:
        """서브클래스가 구현해야 함"""
        pass

    def create_issue(
        self,
        check_name: str,
        severity: IssueSeverity,
        message: str,
        location: Optional[str] = None,
        auto_fix: Optional[str] = None,
        **details
    ) -> ValidationIssue:
        """일관된 ValidationIssue 생성 + 자동 로깅"""
        pass
```

```python
def apply_fixes(
    self,
    app_code: str,
    main_code: str,
    issues: List[ValidationIssue],
) -> tuple[str, str, int]:
    """
    ✅ v0.5.0: Apply auto-fixes to code based on validation issues.

    Returns:
        (fixed_app_code, fixed_main_code, fixes_applied_count)
    """
    fixed_app = app_code
    fixed_main = main_code
    fixes_applied = 0

    for issue in issues:
        if issue.severity not in ("critical", "error"):
            continue  # Only fix critical/error

        try:
            if issue.check_name == "input_widgets":
                fixed_app = self._fix_missing_input_widgets(fixed_app, issue)
                fixes_applied += 1

            elif issue.check_name == "main_call":
                fixed_app = self._fix_main_call(fixed_app, issue)
                fixes_applied += 1

            elif issue.check_name == "input_validation":
                fixed_app = self._fix_validation_pattern(fixed_app, issue)
                fixes_applied += 1

            # ... 5개 fix 메서드
        except Exception as e:
            self.logger.warning(f"Failed to apply fix: {e}")

    return fixed_app, fixed_main, fixes_applied
```

```python
def _fix_missing_input_widgets(self, app_code, issue):
    """
    Insert widget code after "# Input section".
    """
    widget_code = 'keyword = st.text_input("검색 키워드:", key="keyword")'

    pattern = r"(# Input section\s*\n)"
    replacement = rf"\1{widget_code}\n\n"

    return re.sub(pattern, replacement, app_code, count=1)
```

### 핵심 검증: UI Requirements Mapping

```python
def _validate_ui_requirements_mapping(self, tasks, golden_data):
    """
    ✅ This is the KEY check that prevents empty input section!

    Checks:
    - If tasks have {keyword}, inputs are expected
    - If golden_data mentions "입력/검색", inputs should be captured
    """
    issues = []

    # Extract template variables
    import re
    template_vars = set()
    for task in tasks:
        description = task.get("description", "")
        matches = re.findall(r"\{(\w+)\}", description)
        template_vars.update(matches)

    # Check 1: Template vars exist but golden_data doesn't mention them
    if template_vars and golden_data:
        golden_desc = str(golden_data).lower()

        for var_name in template_vars:
            mentioned = (
                var_name.lower() in golden_desc
                or "입력" in golden_desc
                or "input" in golden_desc.lower()
            )

            if not mentioned:
                issues.append(ValidationIssue(
                    severity=ValidationSeverity.WARNING,
                    message=f"Template variable '{var_name}' not mentioned in requirements",
                    auto_fix=f"Add requirement or remove {var_name}"
                ))

    # Check 2: Golden_data mentions input but NO template variables
    if golden_data and not template_vars:
        golden_desc = str(golden_data).lower()

        input_keywords = ["입력", "검색", "키워드", "input", "search", "keyword"]
        mentions_input = any(kw in golden_desc for kw in input_keywords)

        if mentions_input:
            issues.append(ValidationIssue(
                severity=ValidationSeverity.ERROR,
                message="Requirements mention input but no template variables found",
                auto_fix="Add template variables like {keyword} to tasks",
                auto_fix_available=True
            ))

    return issues
```

```python
def _validate_template_variables(self, tasks):
    """
    Check for inconsistent variable names.

    Example: {keyword}, {query}, {search} → Suggest standardize to {keyword}
    """
    keyword_variants = ["keyword", "query", "search"]
    used_variants = [v for v in keyword_variants if v in template_vars]
```

```python
def _validate_tool_role_compatibility(self, agents):
    """
    Ontology 기반 tool-role 매칭.

    Example: role="Data Analyst", tools=["web_search"]
    → Ontology: web_search is for "researcher", not "analyst" → WARNING
    """
    tool_role_map = {
        "web_search": ["researcher", "analyst", "investigator"],
        "file_read": ["analyst", "developer"],
        # ...
    }

    for agent in agents:
        role = agent.get("role", "").lower()
        tools = agent.get("tools", [])

        for tool_name in tools:
            compatible_roles = tool_role_map.get(tool_name, [])

            if role not in compatible_roles:
                issues.append(ValidationIssue(
                    severity=ValidationSeverity.WARNING,
                    message=f"Tool '{tool_name}' may not suit role '{role}'",
                    auto_fix=f"Consider roles: {compatible_roles}"
                ))
```

```python
    Agent goal과 task description의 키워드 일치도 검사.

    Example:
    - Agent goal: "데이터 분석 및 시각화"
    - Task desc: "웹 크롤링 수행"
    → Keyword overlap < 2 → WARNING (semantic mismatch)
    """
    for task in tasks:
        agent = find_assigned_agent(task, agents)

        agent_keywords = set(agent["goal"].split()) - stop_words
        task_keywords = set(task["description"].split()) - stop_words

                message="Agent goal may not match task description",
```
