class DataCleaningTool(BaseTool):
    """데이터 정제 도구"""
    def _run(self, df: pd.DataFrame):
        # 결측값 처리
        df = df.fillna(df.mean(numeric_only=True))

        # 중복 제거
        df = df.drop_duplicates()

        # 이상값 탐지 (IQR 방법)
        Q1 = df.quantile(0.25)
        Q3 = df.quantile(0.75)
        IQR = Q3 - Q1
        outliers = ((df < (Q1 - 1.5 * IQR)) | (df > (Q3 + 1.5 * IQR)))

        return df, outliers

class StatisticalAnalysisTool(BaseTool):
    """통계 분석 도구"""
    def _run(self, df: pd.DataFrame):
        stats = {
            "count": len(df),
            "mean": df.mean(numeric_only=True).to_dict(),
            "median": df.median(numeric_only=True).to_dict(),
            "std": df.std(numeric_only=True).to_dict(),
            "min": df.min(numeric_only=True).to_dict(),
            "max": df.max(numeric_only=True).to_dict(),
        }
        return stats

class ChartGeneratorTool(BaseTool):
    """차트 생성 도구"""
    def _run(self, df: pd.DataFrame, chart_type: str):
        import plotly.express as px

        if chart_type == "line":
            fig = px.line(df, x="date", y="sales", title="월별 매출 추이")
        elif chart_type == "bar":
            fig = px.bar(df, x="region", y="sales", title="지역별 매출")
        elif chart_type == "pie":
            fig = px.pie(df, names="category", values="sales", title="카테고리별 비중")
        elif chart_type == "heatmap":
            corr = df.corr(numeric_only=True)
            fig = px.imshow(corr, title="상관관계 매트릭스")

        return fig

class ReportGeneratorTool(BaseTool):
    """리포트 생성 도구"""
    def _run(self, stats: dict, charts: list, insights: str):
        from reportlab.lib.pagesizes import letter
        from reportlab.pdfgen import canvas

        pdf = canvas.Canvas("report.pdf", pagesize=letter)

        # 제목
        pdf.setFont("Helvetica-Bold", 20)
        pdf.drawString(100, 750, "Sales Analysis Report")

        # 통계 요약
        pdf.setFont("Helvetica", 12)
        y = 700
        for key, value in stats.items():
            pdf.drawString(100, y, f"{key}: {value}")
            y -= 20

        # 인사이트 (LLM 생성)
        pdf.drawString(100, y - 20, "Key Insights:")
        pdf.drawString(120, y - 40, insights)

        pdf.save()
        return "report.pdf"
```

# tools.py
class SalesPredictionTool(BaseTool):
    """판매 예측 도구"""
    def _run(self, df: pd.DataFrame, forecast_periods: int = 30):
        from sklearn.linear_model import LinearRegression

        # 시계열 데이터 준비
        df["date"] = pd.to_datetime(df["date"])
        df = df.sort_values("date")
        df["day_num"] = (df["date"] - df["date"].min()).dt.days

        # 선형 회귀 모델
        X = df[["day_num"]].values
        y = df["sales"].values

        model = LinearRegression()
        model.fit(X, y)

        # 미래 예측
        last_day = df["day_num"].max()
        future_days = np.arange(last_day + 1, last_day + forecast_periods + 1)
        predictions = model.predict(future_days.reshape(-1, 1))

        # 신뢰 구간 계산 (95%)
        residuals = y - model.predict(X)
        std_error = np.std(residuals)
        confidence_interval = 1.96 * std_error

        return {
            "predictions": predictions.tolist(),
            "confidence_interval": confidence_interval,
            "r2_score": model.score(X, y)
        }
```

# tools.py - 새로운 Tool
class ClusteringTool(BaseTool):
    """고객 세분화 도구 (K-Means)"""
    def _run(self, df: pd.DataFrame, n_clusters: int = 3):
        from sklearn.cluster import KMeans

        # RFM 분석 (Recency, Frequency, Monetary)
        rfm = df.groupby("customer_id").agg({
            "date": lambda x: (pd.Timestamp.now() - pd.to_datetime(x).max()).days,  # Recency
            "customer_id": "count",  # Frequency
            "sales": "sum"  # Monetary
        })
        rfm.columns = ["recency", "frequency", "monetary"]

        # K-Means 클러스터링
        kmeans = KMeans(n_clusters=n_clusters, random_state=42)
        rfm["cluster"] = kmeans.fit_predict(rfm)

        # 클러스터 특성
        cluster_summary = rfm.groupby("cluster").mean()

        return rfm, cluster_summary
```

# tools.py - 청크 기반 처리
class DataCleaningTool(BaseTool):
    def _run(self, file_path: str, chunk_size: int = 10000):
        import pandas as pd

        cleaned_chunks = []

        # 청크 단위로 읽기 (메모리 절약)
        for chunk in pd.read_csv(file_path, chunksize=chunk_size):
            chunk = self._clean_chunk(chunk)
            cleaned_chunks.append(chunk)

        # 병합
        df = pd.concat(cleaned_chunks, ignore_index=True)
        return df
```

from concurrent.futures import ThreadPoolExecutor

def parallel_analysis(df: pd.DataFrame):
    with ThreadPoolExecutor(max_workers=4) as executor:
        # 4개 Agent 병렬 실행
        future_clean = executor.submit(agent_1.execute, df)

        # 결과 수집
        cleaned_df = future_clean.result()

        charts = future_viz.result()
        report = future_report.result()
