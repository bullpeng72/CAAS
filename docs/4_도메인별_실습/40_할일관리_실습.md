```python
class TodoCRUDTool(BaseTool):
    """할일 CRUD 작업 도구

    기능:
    - CREATE: 새로운 할일 생성
    - READ: 할일 조회 (ID/전체)
    - UPDATE: 할일 수정
    - DELETE: 할일 삭제
    """
    name: str = "할일_CRUD_도구"
    description: str = "할일을 생성, 조회, 수정, 삭제합니다."

    def _run(self, operation: str, **kwargs) -> dict:
        """CRUD 작업 실행

        Args:
            operation: CREATE, READ, UPDATE, DELETE
            **kwargs: 작업별 파라미터
                - CREATE: title, priority, due_date
                - READ: todo_id (optional)
                - UPDATE: todo_id, title, priority, due_date
                - DELETE: todo_id

        Returns:
            작업 결과 딕셔너리
        """
        if operation == "CREATE":
            return self._create_todo(**kwargs)
        elif operation == "READ":
            return self._read_todos(**kwargs)
        elif operation == "UPDATE":
            return self._update_todo(**kwargs)
        elif operation == "DELETE":
            return self._delete_todo(**kwargs)
        else:
            raise ValueError(f"Unknown operation: {operation}")

    def _create_todo(self, title, priority, due_date):
        """SQLite DB에 할일 저장"""
        import sqlite3
        conn = sqlite3.connect("todos.db")
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO todos (title, priority, due_date, status)
            VALUES (?, ?, ?, 'pending')
        """, (title, priority, due_date))
        conn.commit()
        todo_id = cursor.lastrowid
        conn.close()
        return {"id": todo_id, "status": "created"}

class PriorityAnalysisTool(BaseTool):
    """우선순위 자동 분석 도구

    마감일 기반으로 우선순위를 재계산합니다.
    - 24시간 이내: HIGH
    - 3일 이내: MEDIUM
    - 그 외: LOW
    """
    name: str = "우선순위_분석_도구"
    description: str = "마감일 기반으로 할일의 우선순위를 자동 계산합니다."

    def _run(self, todo_id: int, due_date: str) -> dict:
        """우선순위 계산 로직"""
        from datetime import datetime, timedelta

        due = datetime.strptime(due_date, "%Y-%m-%d")
        now = datetime.now()
        days_left = (due - now).days

        if days_left <= 1:
            priority = "HIGH"
            urgency_score = 10.0
        elif days_left <= 3:
            priority = "MEDIUM"
            urgency_score = 5.0
        else:
            priority = "LOW"
            urgency_score = 2.0

        return {
            "todo_id": todo_id,
            "calculated_priority": priority,
            "days_left": days_left,
            "urgency_score": urgency_score,
            "recommendation": f"{days_left - 1}일 후 시작 권장"
        }

class InputValidationTool(BaseTool):
    """입력 검증 도구"""
    name: str = "입력_검증_도구"
    description: str = "사용자 입력의 유효성을 검사합니다."

    def _run(self, title: str, due_date: str) -> dict:
        """입력 검증"""
        errors = []

        # 제목 검증
        if not title or len(title.strip()) == 0:
            errors.append("제목은 필수입니다")
        elif len(title) > 100:
            errors.append("제목은 100자 이내여야 합니다")

        # 마감일 검증
        try:
            from datetime import datetime
            due = datetime.strptime(due_date, "%Y-%m-%d")
            if due < datetime.now():
                errors.append("마감일은 현재 시간 이후여야 합니다")
        except ValueError:
            errors.append("마감일 형식이 올바르지 않습니다 (YYYY-MM-DD)")

        return {
            "valid": len(errors) == 0,
            "errors": errors
        }
```

```python
# cache.py
import redis
import json

class TodoCache:
    """할일 데이터 캐싱"""
    def __init__(self):
        self.redis = redis.Redis(host='localhost', port=6379, db=0)
        self.ttl = 300  # 5분

    def get_todo(self, todo_id):
        """캐시에서 할일 조회"""
        key = f"todo:{todo_id}"
        data = self.redis.get(key)
        if data:
            return json.loads(data)
        return None

    def set_todo(self, todo_id, todo_data):
        """캐시에 할일 저장"""
        key = f"todo:{todo_id}"
        self.redis.setex(key, self.ttl, json.dumps(todo_data))

    def invalidate(self, todo_id):
        """캐시 무효화"""
        key = f"todo:{todo_id}"
        self.redis.delete(key)
```

**tools.py 수정** (캐싱 적용):
```python
class TodoCRUDTool(BaseTool):
    def __init__(self):
        super().__init__()
        self.cache = TodoCache()

    def _read_todos(self, todo_id=None):
        # 캐시 확인
        if todo_id:
            cached = self.cache.get_todo(todo_id)
            if cached:
                return cached

        # DB 조회
        result = self._query_db(todo_id)

        # 캐시 저장
        if todo_id and result:
            self.cache.set_todo(todo_id, result)

        return result

    def _update_todo(self, todo_id, **kwargs):
        # DB 업데이트
        result = self._execute_update(todo_id, **kwargs)

        # 캐시 무효화
        self.cache.invalidate(todo_id)

        return result
```

**효과**: 반복 조회 **100배 향상** (100ms → 1ms)

```python
class NotificationTool(BaseTool):

        return {"task_id": task.id, "status": "queued"}
```

**효과**: 사용자 대기 시간 **0초** (백그라운드 처리)

**원인**: API 호출 한도 초과

**해결책 1**: Retry 로직 추가
```python
# agents.py
from tenacity import retry, wait_exponential, stop_after_attempt

@retry(wait=wait_exponential(min=1, max=60), stop=stop_after_attempt(5))
def create_agent_with_retry():
    return Agent(...)
```

**해결책 2**: API 키 로테이션
```python
# config.py
import os

API_KEYS = [
    os.environ.get("OPENAI_API_KEY_1"),
    os.environ.get("OPENAI_API_KEY_2"),
    os.environ.get("OPENAI_API_KEY_3"),
]

current_key_index = 0

def get_api_key():
    global current_key_index
    key = API_KEYS[current_key_index]
    current_key_index = (current_key_index + 1) % len(API_KEYS)
    return key
```

```python
DB_CONFIG = {
```
